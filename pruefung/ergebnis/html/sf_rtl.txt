            // translate: replaces placeholders in sourceText by language specific text (i18n)

            //	'firstname': {
            //		'en_US': 'First name',
            //		'de_DE': 'Vorname'
            //	},

            //	}

            // forceLanguage is optional; is forceLanguage is set, this language will be displayed and the user's selection ignored (e.g. 'de_DE')
            this.translate = function (sourceText, i18n, forceLanguage) {

                    targetText = targetText.replace(regExp, newText);

        bertelsmann.customKeywordSearch = new (function () {
            var that = this;
            this.objReplacer = {};

            this.replaceSearchTerm = function (event) {
                jqSearchField = $('form[name="keywordsearch"] input[name="q"]');
                if (!jqSearchField.length) return;

                $.each(jqSearchField, function (idx, searchField) {
                    var keywords = $(searchField).val();

                    var lng = bertelsmann.language.getUserLanguage();
                    if (!that.objReplacer[lng]) return; // if current language has not been configured, exit (nothing to replace)

                    $.each(that.objReplacer[lng], function (idx, objReplacerGroup) {
                        var replaceTo = objReplacerGroup['to'];

                        $.each(objReplacerGroup['from'], function (idx, replaceFrom) {
                            // escape regexp special characters in search string
                            searchStr = replaceFrom.replace(/[-\/\\^$*+?.()|[\]{}]/g, '\\$&');
                            var regExpFrom = new RegExp('(^| )' + searchStr + '($| )', 'gi');
                            keywords = keywords.replace(regExpFrom, '$1' + replaceTo + '$2');
                        });
                    });

                    $(searchField).val(keywords);
                });
            };

            this.add = function (objParam) {
                $.each(objParam, function (idxLng, objLngReplacer) {
                    if (!that.objReplacer[idxLng]) {
                        that.objReplacer[idxLng] = [];
                    }
                    that.objReplacer[idxLng].push(JSON.parse(JSON.stringify(objLngReplacer)));
                });
            };

            this.activate = function (objParam) {
                // we expect the form to not have a submit event; if this ever changes, the standard submit has to be executed after this custom submit
                var jqSearchForm = $('form[name="keywordsearch"]');
                if (jqSearchForm.length === 0) {
                    self.setTimeout(that.activate, 300);
                } else {
                    jqSearchForm.submit(function (event) {
                        that.replaceSearchTerm(event);
                    });
                }
            };
